# ---------------------------------------------------------------- kernel (date-centred) variants
def fit_profiles_k(vals, index, dtype, use, W):
    """Like fit_profiles, but cell (m, day-type, hour) = mean over days weighted by W[m] (a (12, N) kernel weight);
    day-type cells and the 2-day shrinkage towards (wd profile x weekend ratio) are as before, with weighted counts."""
    hr = index.hour.values
    prof = np.full((12, 3, 24), np.nan)
    ok_v = np.isfinite(vals)
    for m in range(12):
        w = np.where(use & ok_v, W[m], 0.0)
        mm = w > 0
        if not mm.any():
            continue
        wd = mm & (dtype == 0)
        if wd.sum() < 24:                                       # no working days: fall back to all days
            wd = mm
        cnt = np.bincount(hr[wd], weights=w[wd], minlength=24)
        wd_prof = np.bincount(hr[wd], weights=vals[wd] * w[wd], minlength=24) / np.maximum(cnt, 1e-12)
        wd_prof[cnt <= 0] = np.nan
        tot0 = np.average(vals[wd], weights=w[wd])
        prof[m, 0] = wd_prof
        for d in (1, 2):
            dd = mm & (dtype == d)
            cnt_d = np.bincount(hr[dd], weights=w[dd], minlength=24)
            cell = np.bincount(hr[dd], weights=vals[dd] * w[dd], minlength=24) / np.maximum(cnt_d, 1e-12)
            cell[cnt_d <= 0] = np.nan
            ratio = (np.average(vals[dd], weights=w[dd]) / tot0) if dd.sum() and np.isfinite(tot0) and tot0 else 1.0
            prior = wd_prof * ratio
            with np.errstate(invalid="ignore"):
                prof[m, d] = np.where(cnt_d > 0, (cnt_d * cell + 2 * prior) / (cnt_d + 2), prior)
    return prof


def monthly_kbar_k(pv, clear, use, W):
    """Kernel version of monthly_kbar: weighted sum(pv) / sum(clear) per target month (NaN if no weight)."""
    kbar = np.full(12, np.nan)
    for m in range(12):
        w = np.where(use & (clear > ah.CS_MIN_KW), W[m], 0.0)
        if (w * clear).sum() > 0:
            kbar[m] = (w * pv).sum() / (w * clear).sum()
    return kbar


def _fill_nov_kbar(kbar):
    if not np.isfinite(kbar[10]):
        kbar[10] = np.nanmean([kbar[9], kbar[11]])
    return kbar
